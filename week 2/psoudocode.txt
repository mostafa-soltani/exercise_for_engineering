def f(x):
    retrun(x ** x - (10*x) - 20)


def find_range():
    last_num = 0
    current_num = 1

    if f(current_num) not positive:
        last_num = current_num
        current_num = current_num * 2

    elif f(current_num) is positive:
        return last_num , current_num


def binary(low,high):
    low,high = find_range()

    while low <= high:
        mid = low + (high - low) / 2

        if (f(mid) > 0 and (mid == low or f(mid -1) <= 0)):
            return mid

        if (f(mid) <= 0):
            return binary((mid +1),high)

        else:
            return binary(low,(mid -1))
